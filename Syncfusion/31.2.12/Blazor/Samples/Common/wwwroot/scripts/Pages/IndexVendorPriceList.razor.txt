    private async Task LoadAssembliesAsync(bool clearTree = true, string? parentWhereClause = null, int currentLevel = -1){
        // 1. Preparation & state management
        if (clearTree){
            TreeNodes.Clear();
        }
        Console.WriteLine($"LoadAssembliesAsync Index={CurrentViewIndex}");
        var currentView = Views[CurrentViewIndex];
        var totalLevels = ParseCount(currentView.DisplayFields); // equivalent to levels = Parse(...)

        // 2. Determine current level we're loading
        var level = currentLevel;

        // If we're at the deepest level → nothing more to load
        if (level + 1 >= totalLevels){
            return;
        }

        // 3. Get field names for CURRENT level we're about to load
        var keyField = ParseField(currentView.KeyFields, level + 1); // Level + 2 - 1 in 0-based
        var displayField = ParseField(currentView.DisplayFields, level + 1);
        Console.WriteLine($"LoadAssembliesAsync keyField={keyField}");

        // 4. Build SQL query (parameterized!)
        var sql = $@"
        SELECT DISTINCT 
            {keyField} AS KeyValue,
            {displayField} AS DisplayValue
        FROM VendorPrices
        WHERE DivisionID IN (0, @DivisionID)";
        if (!string.IsNullOrWhiteSpace(parentWhereClause)){
            sql += $"\nAND {parentWhereClause}";
        }
        //sql += "\nORDER BY DisplayValue";
        LastViewQuery = sql; // for debugging

        // Execute query using your DB wrapper
        var parameters = new { DivisionID = 1 }; // ← use your real source
        var dt = db.SqlExec(sql, parameters); // assuming returns DataTable

        // 5. Process results
        var newNodes = new List<TreeNodeModel>();
        foreach (DataRow row in dt.Rows){
            var keyValue = row["KeyValue"]?.ToString()?.Trim() ?? "";
            var displayValue = row["DisplayValue"]?.ToString()?.Trim() ?? "";

            // Special display value transformations (exact VB6 logic)
            if (displayValue == "- global -")
                displayValue = $"Global (any {CD_Community})";
            if (displayValue == "- corporate -")
                displayValue = "Corporate (any Division)";
            var isSpecial = IsSpecialNode(displayValue);
            var node = new TreeNodeModel {
                Name = displayValue,
                KeyFieldName = keyField,
                DisplayFieldName = displayField,
                WhereClause = string.IsNullOrEmpty(parentWhereClause)
                    ? $"[{keyField}] = {DbQuote(keyValue)}"
                    : $"{parentWhereClause} AND [{keyField}] = {DbQuote(keyValue)}",
                IsGlobalOrSpecial = isSpecial,
                HasChildren = level + 2 < totalLevels // will have children if not deepest level
            };

            // For lazy loading: if it can have children, we mark it as having children
            // Actual children will be loaded when expanded
            if (node.HasChildren){
                // We don't load children yet → just mark it expandable
                node.Children = new List<TreeNodeModel>(); // empty → SfTreeView shows expand icon
            }
            newNodes.Add(node);
        }

        // 6. Attach nodes to correct parent
        if (level == -1){
            // Root level
            TreeNodes.AddRange(newNodes);
        }
        else{
            // Find parent node and attach children
            // In real app you usually keep reference to parent node
            // Here we demonstrate simple approach - in practice use better tracking
            // (this part is the most different from VB6 grid)
            // You typically track expanded node in event handler
        }
        await InvokeAsync(StateHasChanged);
    }

    private static int ParseCount(string input){
        return string.IsNullOrEmpty(input) ? 0 : input.Split(',').Length;
    }

    private static string ParseField(string input, int index) // 1-based
    {
        if (string.IsNullOrEmpty(input)) return "";
        var parts = input.Split(',');
        return index >= 0 && index <= parts.Length ? parts[index].Trim() : "";
    }

    private static bool IsSpecialNode(string displayValue){
        var CD_Community = "";
        var special = new[] {
            $"- any {CD_Community} -",
            "- any phase -",
            "- any assembly -",
            $"Global (any {CD_Community})",
            "Corporate (any Division)"
        };
        return special.Contains(displayValue);
    }

    private string DbQuote(string? value){
        if (value is null) return "NULL";
        return $"'{value.Replace("'", "''")}'";

                    displayValue = $"Global (any {CD_Community})";
                nodeLevel0 = new TreeNodeModel {
                    Name = displayValue,
                    WhereClause = $"{level0ID}='{level0}'",
                    //$"AND {currentView.KeyFields.Split(',')[0].Trim()} = {DbQuote(item["Column2"])}",
                    IsGlobalOrCorporate = true
                };
                TreeNodes.Add(nodeLevel0);
                prevLevel0 = level0;
                prevLevel1 = Init;
                prevLevel2 = Init;
                prevLevel3 = Init;
            }
            var level1 = item[level1ID]?.ToString();
            if (level1 != prevLevel1){
                if (true || prevLevel1 != Init){
                    nodeLevel1.Children = level1Children;
                    level1Children = new List<TreeNodeModel>(0);
                    ;
                }
                nodeLevel1 = new TreeNodeModel {
                    Name = item[level1ID + "Display"]?.ToString(),
                    WhereClause = $"{level0ID}='{level0}' AND {level1ID}='{level1}'",
                    //$"AND {currentView.KeyFields.Split(',')[0].Trim()} = {DbQuote(item["Column2"])}",
                    IsGlobalOrCorporate = true
                };
                prevLevel1 = level1;
                prevLevel2 = Init;
                prevLevel3 = Init;
                level0Children.Add(nodeLevel1);
            }
            var level2 = item[level2ID]?.ToString();
            if (level2 != prevLevel2){
                if (true || prevLevel2 != Init){
                    nodeLevel2.Children = level2Children;
                    level2Children = new List<TreeNodeModel>(0);
                    ;
                }
                nodeLevel2 = new TreeNodeModel {
                    Name = item[level2ID + "Display"]?.ToString(),
                    WhereClause = $"{level0ID}='{level0}' AND {level1ID}='{level1}' AND {level2ID}='{level2}'",
                    //$"AND {currentView.KeyFields.Split(',')[0].Trim()} = {DbQuote(item["Column2"])}",
                    IsGlobalOrCorporate = true
                };
                prevLevel2 = level2;
                prevLevel3 = Init;
                level1Children.Add(nodeLevel2);
            }
            var level3 = item[level3ID]?.ToString();
            if (level3 != prevLevel3){
                if (prevLevel3 != Init){
                    //nodeLevel3.Children = level3Children;
                    //level3Children = new List<TreeNodeModel>(0);;
                }
                nodeLevel3 = new TreeNodeModel {
                    Name = item[level3ID + "Display"]?.ToString(),
                    WhereClause = $"{level0ID}='{level0}' AND {level1ID}='{level1}' AND {level2ID}='{level2}' AND {level3ID}='{level3}'",
                    //$"AND {currentView.KeyFields.Split(',')[0].Trim()} = {DbQuote(item["Column2"])}",
                    IsGlobalOrCorporate = true
                };
                prevLevel3 = level3;
                level2Children.Add(nodeLevel3);
            }

            // Add dummy child so we know it can expand

            //nodeLevel0.Children = new List<TreeNodeModel> { new TreeNodeModel { Name = "Loading..." } };
        }
        nodeLevel2.Children = level2Children;
        nodeLevel1.Children = level1Children;
        nodeLevel0.Children = level0Children;
        StateHasChanged();
    }

    private string DbQuote(object value){
        if (value == null) return "NULL";
        if (value is string s) return $"'{s.Replace("'", "''")}'";
        if (value is DateTime dt) return $"'{dt:yyyy-MM-dd}'";
        return value.ToString();
    }

            FROM VendorPrices v
            LEFT OUTER JOIN PriceGroups g 
                ON v.PriceLink = g.PriceGroup 
               AND g.DivisionID = @DivisionID";
            string? whereClause = null;
