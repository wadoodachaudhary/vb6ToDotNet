    private async Task LoadJobDataAsync(){
        string sql = @"
            SELECT wj.* 
            FROM workflowjobs wj
            JOIN tbljobs j ON (wj.DivisionID = j.DivisionID AND wj.job = j.job_no)
            LEFT OUTER JOIN DivisionCommunities d ON (d.Community = j.Community)
            WHERE (d.DivisionID IS NULL OR d.DivisionID = @DivisionID)
            AND j.DivisionID = @DivisionID
            ORDER BY wj.job";
        var parameters = new { DivisionID = 1 }; // Replace with actual HFApp.DivisionID equivalent
        var result = await db.QueryAsync(sql, parameters);
        // Map results to view model
        foreach (var row in result){
            var item = new Job() {
                JobNumber = row["Job"]?.ToString() ?? "",
                JobDescription = row["Description"]?.ToString() ?? "",

        await JobsGrid.Refresh();
