    private async Task LoadDataAsync(){
        if (string.IsNullOrEmpty(mVendor)){
            await ClearFormAsync();
            return;
        }

        // ─────────────────────────────────────────────────────────────
        // Exact same query structure - only using parameters
        // ─────────────────────────────────────────────────────────────
        var sql = @"
            SELECT wp.walletid AS WalletPayeeID, v.*
            FROM tblVendors v
            JOIN divisions d ON v.divisionid = d.divisionid
            LEFT JOIN datasources ds ON d.bookofaccount = ds.bookofaccount
            LEFT JOIN wallet_payees wp ON wp.datasourceid = ds.datasourceid 
                                      AND wp.payeeid = v.vendor_id
            WHERE v.Vendor_ID = @VendorId
              AND v.divisionid = @DivisionId";

        // sql = @"
        //     SELECT v.*
        //     FROM tblVendors v
        //     WHERE v.Vendor_ID = @VendorId
        //       AND v.divisionid = @DivisionId";
        var parameters = new {
            VendorId = mVendor,
            DivisionId = 1 // ← replace with your actual source
        };
        var results = await db.QueryAsync(sql, parameters);
        if (!results.Any()){
            await ClearFormAsync();
            return;
        }
        var row = results.First();
        IsReadOnly = false;

        // ─────────────────────────────────────────────────────────────
        // Map all fields exactly as in original
        // ─────────────────────────────────────────────────────────────
        VendorId = row["Vendor_ID"]?.ToString() ?? "";
        Description = row["Vendor_Name"]?.ToString() ?? "";
        VendorDesc = $"{Description} ({VendorId})";
        Address = row["Addr1"]?.ToString() ?? "";
        City = row["city"]?.ToString() ?? "";
        Province = row["state"]?.ToString() ?? "";
        Postal = row["zip"]?.ToString() ?? "";
        Phone = row["phone"]?.ToString() ?? "";
        Fax = row["fax"]?.ToString() ?? "";
        IsTBD = row["isTBD"]?.ToString()?.Equals("true", StringComparison.OrdinalIgnoreCase) == true;
        BuildProEnabled = row["BuildProEnabled"]?.ToString()?.Equals("true", StringComparison.OrdinalIgnoreCase) == true;
        WebUID = row["WebUID"]?.ToString() ?? "";
        WebPWD = row["password"]?.ToString() ?? "";
        WalletPayeeID = row["WalletPayeeID"]?.ToString() ?? "";
        TaxID = row["TaxID"]?.ToString() ?? "";
        StatusIndex = row["Inactive"]?.ToString()?.Equals("True", StringComparison.OrdinalIgnoreCase) == true ? 1 : 0;
        TradeType = row["VendorGroupID"]?.ToString() ?? "";
        POFormat = row["POFormat"]?.ToString() ?? "";
        HoldbackPercent = (double)row["HoldbackPercentage"];

        // ─────────────────────────────────────────────────────────────
        // Insurance grid mapping
        // ─────────────────────────────────────────────────────────────
        InsuranceData = new List<InsuranceRequirement> {
            new() {
                Type = "General", Required = (bool)row["glinsrequired"],
                ExpiryDate = Convert.ToDateTime(row["glinsexpdate"]),
                Company = row["GLInsCompany"]?.ToString(),
                PolicyNumber = row["GLInsPolicyNumber"]?.ToString()
            },
            new() {
                Type = "Workers", Required = (bool)row["wcinsrequired"],
                ExpiryDate = Convert.ToDateTime(row["wcinsexpdate"]),
                Company = row["WCInsCompany"]?.ToString(),
                PolicyNumber = row["WCInsPolicyNumber"]?.ToString()
            },
            new() {
                Type = "Umbrella", Required = (bool)row["umbinsrequired"],
                ExpiryDate = Convert.ToDateTime(row["umbinsexpdate"]),
                Company = row["UmbInsCompany"]?.ToString(),
                PolicyNumber = row["UmbInsPolicyNumber"]?.ToString()
            },
            new() {
                Type = "Automobile", Required = (bool)row["autoinsrequired"],
                ExpiryDate = Convert.ToDateTime(row["autoinsexpdate"]),
                Company = row["AutoInsCompany"]?.ToString(),
                PolicyNumber = row["AutoInsPolicyNumber"]?.ToString()
            }
        };
        await InsuranceGrid?.Refresh()!;

        // ─────────────────────────────────────────────────────────────
        // Tax Groups - Default row + Community specific
        // ─────────────────────────────────────────────────────────────
        TaxGroupsData.Clear();

        // Default vendor tax groups row
        TaxGroupsData.Add(new TaxGroupRow {
            Community = "",
            CommunityDescription = " -- vendor defaults --",
            Labour = row["LabourTaxGroup"]?.ToString(),
            Material = row["MaterialTaxGroup"]?.ToString(),
            SubContract = row["SubContractTaxGroup"]?.ToString(),
            Equipment = row["EquipmentTaxGroup"]?.ToString(),
            Overhead = row["OverheadTaxGroup"]?.ToString(),
            Other = row["OtherTaxGroup"]?.ToString()
        });

        // Community-specific tax groups
        var taxSql = @"
            SELECT t.community, c.description,
                   t.MaterialTaxGroup, t.LabourTaxGroup,
                   t.SubContractTaxGroup, t.EquipmentTaxGroup,
                   t.OverheadTaxGroup, t.OtherTaxGroup
            FROM vendortaxgroups t
            JOIN tbllocality c ON t.Community = c.area
            WHERE t.vendor = @Vendor";
        var taxParams = new { Vendor = mVendor };
        var taxResults = await db.QueryAsync(taxSql, taxParams);
        foreach (var taxRow in taxResults){
            TaxGroupsData.Add(new TaxGroupRow {
                Community = taxRow["community"]?.ToString(),
                CommunityDescription = taxRow["description"]?.ToString(),
                Material = taxRow["MaterialTaxGroup"]?.ToString(),

                SubContract = taxRow["SubContractTaxGroup"]?.ToString(),
                Equipment = taxRow["EquipmentTaxGroup"]?.ToString(),
                Overhead = taxRow["OverheadTaxGroup"]?.ToString(),
                Other = taxRow["OtherTaxGroup"]?.ToString()
            });
        }

        // Empty new row

    private async Task LoadPayPointsAsync(){
        var sql = $@"
            SELECT * 
            FROM VendorPaymentPoints 
            WHERE DivisionID = @DivisionID 
              AND Vendor = @Vendor 
            ORDER BY POIndex";
        var parameters = new { DivisionID = DivisionID, Vendor = mVendor };
        var results = await db.QueryAsync(sql, parameters);
        PayPoints.Clear();
        foreach (var row in results){
            var payPoint = new VendorPayPoint {
                POIndex = row["POIndex"]?.ToString() ?? "",
                PayPoint1Percent = string.IsNullOrEmpty(row["PayPoint1Percent"]?.ToString()) ? 0 : Convert.ToDouble(row["PayPoint1Percent"]),
                PayPoint2Percent = string.IsNullOrEmpty(row["PayPoint2Percent"]?.ToString()) ? 0 : Convert.ToDouble(row["PayPoint2Percent"]),
                PayPoint3Percent = string.IsNullOrEmpty(row["PayPoint3Percent"]?.ToString()) ? 0 : Convert.ToDouble(row["PayPoint3Percent"]),
                PayPoint4Percent = string.IsNullOrEmpty(row["PayPoint4Percent"]?.ToString()) ? 0 : Convert.ToDouble(row["PayPoint4Percent"]),
                PayPoint5Percent = string.IsNullOrEmpty(row["PayPoint5Percent"]?.ToString()) ? 0 : Convert.ToDouble(row["PayPoint5Percent"])
            };
            PayPoints.Add(payPoint);
        }
        PayPoints.Add(new VendorPayPoint()); // Empty row
        await PayPointsGrid.Refresh();
    }

    private async Task LoadContactsAsync(){
        var sql = $@"

            new { DivisionID = DivisionID });
        if (vendorData.Count > 0){
            foreach (var row in vendorData){
                vendors.Add(row["Vendor_ID"]?.ToString() ?? "");
            }
        }
        VendorId = vendors[VendorIndex];
        mVendor = VendorId;
        Console.WriteLine($" OnInitializedAsync {VendorId}");
        LoadDataAsync();
    }

    private async Task LoadPayPoints(){
        var sql = @"SELECT * FROM VendorPaymentPoints 
                    WHERE DivisionID = @DivisionID AND Vendor = @Vendor 
                    ORDER BY POIndex";
        var results = await db.QueryAsync(sql, new { DivisionID = DivisionID, Vendor = VendorId });
        PayPoints.Clear();
        foreach (var row in results){
            PayPoints.Add(new VendorPayPoint {
                POIndex = row["POIndex"]?.ToString(),
                PayPoint1Percent = Convert.ToDouble(row["PayPoint1Percent"] ?? 0),
                PayPoint2Percent = Convert.ToDouble(row["PayPoint2Percent"] ?? 0),
                PayPoint3Percent = Convert.ToDouble(row["PayPoint3Percent"] ?? 0),
                PayPoint4Percent = Convert.ToDouble(row["PayPoint4Percent"] ?? 0),
                PayPoint5Percent = Convert.ToDouble(row["PayPoint5Percent"] ?? 0)
            });
        }
        PayPoints.Add(new VendorPayPoint()); // Empty row like AddItem ""
    }

    private async Task LoadContacts(){
        var sql = @"SELECT * FROM contacts 
                    WHERE DivisionID = @DivisionID AND contacttypeid=99 
                    AND vendorcode = @Vendor 
                    ORDER BY firstname";
