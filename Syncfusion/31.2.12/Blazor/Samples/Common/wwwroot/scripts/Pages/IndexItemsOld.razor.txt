    private void SafeSe2t(IDictionary<string, object> dict, string key, object value){
        if (value == DBNull.Value || value == null)
            dict[key] = "";
        else
            dict[key] = value;
    }

// Helper to safely set values (handles null/DBNull)
    private void SafeSet(IDictionary<string, object> dict, string key){
        object value = dict[key];
        if (value == DBNull.Value || value == null)
            dict[key] = "";
        else
            dict[key] = value;
    }

// Phase selection handler
    private async Task OnPhaseSelected(RowSelectEventArgs<ExpandoObject> args){
        if (mDirty){
// Prompt to save changes (similar to VB6)
            var confirm = await JSRuntime.InvokeAsync<bool>("confirm", "Unsaved changes. Save before switching?");
            if (confirm){
                await SaveItemsAsync();
            }
        }
        dynamic selected = args.Data;
        SelectedPhase = selected.Phase;
        await LoadItemsForPhaseAsync(SelectedPhase);
        mDirty = false;
    }

// Save changes (Batch edit mode saves on Update)
    private async Task SaveItemsAsync(){
// End batch edit
        await ItemsGrid.EndEditAsync();

// Save each modified/deleted/new item
        foreach (dynamic item in ItemDataList){
            if (item.IsDeleted){
                await db.ExecuteAsync(@"
                    DELETE FROM tblItems 
                    WHERE Phase = @Phase AND Item = @Item",
                    new { Phase = item.Phase, Item = item.Item });
            }
            else if (item.IsNew){
// Build INSERT sql dynamically, excluding flags
                var fields = new List<string>();
                var values = new List<string>();
                var parameters = new ExpandoObject() as IDictionary<string, object>;
                foreach (var kv in (IDictionary<string, object>)item){
                    if (kv.Key != "IsDirty" && kv.Key != "IsDeleted" && kv.Key != "IsNew"){
                        fields.Add(kv.Key);
                        values.Add("@" + kv.Key);
                        parameters.Add(kv.Key, kv.Value);
                    }
                }
                var sql = $"INSERT INTO tblItems ({string.Join(",", fields)}) VALUES ({string.Join(",", values)})";
                await db.ExecuteAsync(sql, parameters);
            }
            else if (item.IsDirty){
// Build UPDATE sql dynamically, excluding keys and flags
                var sets = new List<string>();
                var parameters = new ExpandoObject() as IDictionary<string, object>;
                foreach (var kv in (IDictionary<string, object>)item){
                    if (kv.Key != "Phase" && kv.Key != "Item" && kv.Key != "IsDirty" && kv.Key != "IsDeleted" && kv.Key != "IsNew"){
                        sets.Add($"{kv.Key} = @{kv.Key}");
                        parameters.Add(kv.Key, kv.Value);
                    }
                }
                parameters.Add("Phase", item.Phase);
                parameters.Add("Item", item.Item);
                var sql = $"UPDATE tblItems SET {string.Join(", ", sets)} WHERE Phase = @Phase AND Item = @Item";
                await db.ExecuteAsync(sql, parameters);
            }
        }
        mDirty = false;
        await JSRuntime.InvokeVoidAsync("alert", "Changes saved.");
// Reload to reflect changes
        await LoadItemsForPhaseAsync(SelectedPhase);
    }

// Action handlers (from menus in VB6)
    private async Task OnActionBegin(ActionEventArgs<ExpandoObject> args){
        if (args.RequestType == Syncfusion.Blazor.Grids.Action.Delete){
            dynamic data = args.Data;
            data.IsDeleted = true;
            mDirty = true;
            args.Cancel = true; // Prevent default delete, mark instead
        }
        else if (args.RequestType == Syncfusion.Blazor.Grids.Action.Add){
            dynamic data = args.Data;
            data.IsNew = true;
            mDirty = true;
        }
    }

    private async Task OnActionComplete(ActionEventArgs<ExpandoObject> args){
        if (args.RequestType == Syncfusion.Blazor.Grids.Action.Save){
            dynamic data = args.Data;
            data.IsDirty = true;
            mDirty = true;
        }
    }

// Duplicate Items
    private async Task DuplicateItems(){
        var selected = await ItemsGrid.GetSelectedRecordsAsync();
        if (selected.Any()){
            foreach (dynamic sel in selected){
                dynamic duplicate = new ExpandoObject();
                foreach (var kv in (IDictionary<string, object>)sel){
                    ((IDictionary<string, object>)duplicate)[kv.Key] = kv.Value;
                }
                duplicate.Item = duplicate.Item + "_copy"; // Adjust
                duplicate.IsNew = true;
                ItemDataList.Add(duplicate);
            }
            await ItemsGrid.Refresh();
            mDirty = true;
        }
    }

// Delete Items (mark hidden/deleted)
    private async Task DeleteItems(){
        var selected = await ItemsGrid.GetSelectedRecordsAsync();
        foreach (dynamic item in selected){
            item.IsDeleted = true;
        }
        await ItemsGrid.Refresh();
        mDirty = true;
    }

// PickList handlers
    private Task<ExpandoObject> ShowPickListAsync(){
        var tcs = new TaskCompletionSource<ExpandoObject>();
        PickListCallback = (result) => {
            PickListVisible = false;
            tcs.SetResult(result);
            StateHasChanged();
        };
        return tcs.Task;
    }

    private void OnPickListRowSelected(RowSelectEventArgs<ExpandoObject> args){
        SelectedPickListItem = args.Data;
    }

    private void OnPickListSelect(){
        PickListCallback(SelectedPickListItem);
    }
