        public DataTable SqlExec(string query){
            using (var connection = new SQLiteConnection(connectionString)){
                connection.Open();
                //SQLiteFunctionRegistry.RegisterFunctions(connection);
                using (var command = new SQLiteCommand(query, connection)){

        public async Task<List<Dictionary<string, object>>> QueryAsync(string sql, object parameters = null){
            var results = new List<Dictionary<string, object>>();
            using var connection = new SQLiteConnection(connectionString);
            await connection.OpenAsync();
            using var command = connection.CreateCommand();
            command.CommandText = sql;

            // Add parameters safely
            if (parameters != null){
                foreach (var prop in parameters.GetType().GetProperties()){
                    var value = prop.GetValue(parameters) ?? DBNull.Value;
                    command.Parameters.AddWithValue("@" + prop.Name, value);
                }
            }
            using var reader = await command.ExecuteReaderAsync();
            while (await reader.ReadAsync()){
                var row = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase);
                for (var i = 0; i < reader.FieldCount; i++){
                    var name = reader.GetName(i);
                    var value = reader.IsDBNull(i) ? null : reader.GetValue(i);
                    row[name] = value;
                }
                results.Add(row);
            }
            return results;
        }

        // For INSERT/UPDATE/DELETE
        public async Task<int> ExecuteAsync(string sql, object parameters = null){
            using var connection = new SQLiteConnection(connectionString);
            await connection.OpenAsync();
            using var command = connection.CreateCommand();
            command.CommandText = sql;
            if (parameters != null){
                foreach (var prop in parameters.GetType().GetProperties()){
                    command.Parameters.AddWithValue("@" + prop.Name, prop.GetValue(parameters) ?? DBNull.Value);
                }
            }
            return await command.ExecuteNonQueryAsync();
        }

        public async Task<List<ExpandoObject>> QueryDynamicAsync(string sql, object parameters = null){
            var results = new List<ExpandoObject>();
            using var connection = new SQLiteConnection(connectionString);
            await connection.OpenAsync();
            using var command = connection.CreateCommand();
            command.CommandText = sql;

            // Add parameters if provided
            if (parameters != null){
                foreach (var prop in parameters.GetType().GetProperties()){
                    var value = prop.GetValue(parameters) ?? DBNull.Value;
                    command.Parameters.AddWithValue("@" + prop.Name, value);
                }
            }
            using var reader = await command.ExecuteReaderAsync();
            while (await reader.ReadAsync()){
                var row = new ExpandoObject();
                var dict = (IDictionary<string, object>)row;
                for (var i = 0; i < reader.FieldCount; i++){
                    var columnName = reader.GetName(i);
                    var value = reader.IsDBNull(i) ? null : reader.GetValue(i);
                    dict[columnName] = value;
                }
                results.Add(row);
            }
            return results;
        }

    /// <summary>
    /// C# version of VB6 LoadPricingCommunities()
    /// Populates the pricing community dropdown (cboPriceCommunity)
    /// </summary>
    public async Task LoadPricingCommunitiesAsync(string? limitToCommunity = null){
        //Console.WriteLine($"LoadPricingCommunitiesAsync:");
        var divisionId = Convert.ToInt32(AppState_DivisionID); // your app context
        var limitClause = string.IsNullOrEmpty(limitToCommunity)
            ? ""
            : "AND c.area = @LimitToCommunity";
        var sql = $@"
            -- Blank entry
            SELECT 
                c.Description,
                c.area + CHAR(2) AS Value,
                0 AS SortOrder
            FROM tbllocality c
            LEFT OUTER JOIN DivisionCommunities d ON d.Community = c.Area AND d.DivisionID = @DivisionID
            WHERE ISNULL(c.inactive, 0) = 0
              AND (@DivisionID IS NULL OR d.DivisionID = @DivisionID)
              {limitClause}
            UNION
            -- Communities + Phases
            SELECT 
                c.Description + ISNULL(' ' + p.description, ''),
                c.area + CHAR(2) + ISNULL(p.communityphase, '') AS Value,
                0 AS SortOrder
            FROM tbllocality c
            LEFT OUTER JOIN communityphase p ON c.area = p.community
            LEFT OUTER JOIN DivisionCommunities d ON d.Community = c.Area AND d.DivisionID = @DivisionID
            WHERE ISNULL(c.inactive, 0) = 0
              AND (@DivisionID IS NULL OR d.DivisionID = @DivisionID)
              {limitClause}
            ORDER BY Description";
        var parameters = new {
            DivisionID = divisionId > 0 ? (int?)divisionId : null,
            LimitToCommunity = limitToCommunity
        };
        var rows = await db.QueryAsync(sql, parameters);
        PricingCommunities = rows.Select(row => new CommunityItem {
                PricingCommunityPhase = row["Description"]?.ToString() ?? "",
                PricingCommunity = row["Value"]?.ToString() ?? "",
                SortOrder = Convert.ToInt32(row["SortOrder"])
            })
            .OrderBy(x => x.PricingCommunityPhase)
            .ToList();

        // Optional: Add blank at top
        PricingCommunities.Insert(0, new CommunityItem {
            PricingCommunityPhase = "",
            PricingCommunity = "",
            SortOrder = 0
        });
        StateHasChanged();
    }

    private async Task LoadItemsGridAsync(){

            PricingCommunityPhase = mPricingCommunityPhase.Trim() ?? ""
        });

        AssemblyCost = 0;
        foreach (DataRow row in dt.Rows){
            var item = new AssemblyItemRow();
            item.Sequence = (int)row["sequence"];
            item.Phase = row["Phase"]?.ToString();
            item.Item = row["Item"]?.ToString();
            item.ItemNumber = row["itemnumber"]?.ToString();
            item.ItemChart = row["itemchart"]?.ToString();
            item.Description = row["Description"]?.ToString() ?? "";
            item.TakeoffUOM = row["TakeoffUOM"]?.ToString() ?? "";
            item.OrderUOM = row["OrderUOM"]?.ToString() ?? "";
            item.Formula = row["Formula"]?.ToString() ?? "";
            item.TakeoffQty = ToDouble(row["TakeoffQty"]);
            item.OrderQty = ToDouble(row["OrderQty"]);
            item.WastePercent = ToDouble(row["WastePercent"]);
            item.Notes = row["notes"]?.ToString() ?? "";
            item.Location = row["Location"]?.ToString() ?? "";
            item.IsQuote = row["IsQuote"]?.ToString() == "True";
            item.Invertable = row["Invertable"]?.ToString() == "True" && (int)row["hasInverseItem"] == 1;
            item.UseModelCost = row["UseModelCost"]?.ToString() == "True";
            item.POIndex = row["POIndex"]?.ToString() ?? "";
            item.POIndexDescription = row["POIndexDescription"]?.ToString() ?? "";
            item.JCCostCode = row["JCCostCode"]?.ToString() ?? "";
            item.JCCategory = row["JCCategory"]?.ToString() ?? "";
            // WBS01 to WBS40
            for (var i = 1; i <= 40; i++)
                item.WBS[i] = row[$"WBS{i:D2}"]?.ToString() ?? "";
            if (!string.IsNullOrEmpty(PricingCommunity)){
                var quality = (int)row["RateQuality"];
                item.PriceLevel = quality switch {
                    4 or 5 or 6 => "Quote",
                    9 => "Phase Specific",
                    10 => $"{CommunityTerm} Specific",
                    11 => $"Global (any {CommunityTerm})",
                    12 => "Corporate (any Division)",
                    13 => "Item DB",
                    _ => "Unit Price"
                };
                item.Price = ToDouble(row["Rate"]);
                item.TaxRate = ToDouble(row["JCTaxRate"]) / 100;
                var qty = item.OrderQty;
                item.PretaxAmount = item.Price * qty;
                item.TaxAmount = item.PretaxAmount * item.TaxRate;
                item.ExtendedAmount = item.PretaxAmount + item.TaxAmount;
                AssemblyCost += item.ExtendedAmount;
                item.Vendor = row["Vendor"]?.ToString() ?? "";
                item.VendorDesc = row["VendorDesc"]?.ToString() ?? "";
                item.OldPriceLevel = item.PriceLevel;
            }
            else{
                item.ItemType = item.IsQuote ? "Quote" : "Unit Price";
            }
            items.Add(item);
        }
        ItemsData = items;
        AssemblyCostLabel = AssemblyCost.ToString("$#,##0.00");
    }

    private string BuildComponentsQuery(){
        // This is the EXACT translation of the VB6 component cost query
        // Used when PricingCommunity is set and components are enabled
        return $@"
        SELECT 
            m.assemblyid,
            m.community,
            m.assembly,
            m.model,
            m.optionid,
            m.description,
            c.qty,
            ROUND(SUM(ISNULL(0, 0) * (100 + ISNULL(t.JCRate, 0)) / 100 * ISNULL(d.OrderQty, 0)), 2) AS Cost
        FROM tblDBAssemblyComponents c
        JOIN tblDBAssemblyMaster m 
            ON c.componentassemblyid = m.assemblyid
        LEFT OUTER JOIN tblDBAssemblyDetails d 
            ON m.assemblyid = d.assemblyid
        LEFT OUTER JOIN tblPhaseItem i 
            ON d.DivisionID = i.DivisionID 
            AND d.Phase = i.Phase 
            AND d.Item = i.Item
        LEFT OUTER JOIN tblPOIndex p 
            ON p.DivisionID = @DivisionID 
            AND ISNULL(ISNULL(d.POIndex,''), ISNULL(i.POIndex,'')) = p.POIndex
        LEFT OUTER JOIN StandardCostCodes cc 
            ON cc.DivisionID = @DivisionID 
            AND ISNULL(ISNULL(i.JCCostCode,''), ISNULL(p.JCCostCode,'')) = cc.CostCode
        LEFT OUTER JOIN StandardCategories ct 
            ON i.DivisionID = ct.DivisionID 
            AND i.JCCategory = ct.Category
        LEFT OUTER JOIN tblvendors v 
            ON v.DivisionID = d.DivisionID 
            AND v.vendor_id = dbo.Purch_GetCommunityVendor(@PricingCommunity, p.POIndex, @DivisionID)
        LEFT OUTER JOIN taxGroups t 
            ON t.DivisionID = @DivisionID 
            AND t.TaxGroup = dbo.Purch_GetDefaultTaxGroup(
                '', @PricingCommunity, '', d.Model, d.Assembly, 
                i.phase, i.item, v.Vendor_ID, i.JCCategory, @DivisionID
            )
        WHERE c.ParentAssemblyid = @ParentAssemblyId
        GROUP BY 
            m.assemblyid, m.community, m.assembly, m.model, m.optionid, 
            m.description, c.qty
        ORDER BY m.assembly";
    }

            FROM POAreaVendor
            WHERE POIndex = @POIndex
              AND DivisionID = @DivisionID
              AND (Area = @Community OR Area = '' OR Area IS NULL)
            ORDER BY 
                CASE 
                    WHEN Area = @Community THEN 3
                    WHEN Area = '' OR Area IS NULL THEN 2
                    ELSE 1 
                END DESC,
                CASE WHEN Area IS NULL THEN 1 ELSE 0 END DESC
            LIMIT 1";
        using var command = conn.CreateCommand();
        command.CommandText = sql;
        command.Parameters.AddWithValue("@Community", community);
        command.Parameters.AddWithValue("@POIndex", poIndex);
        command.Parameters.AddWithValue("@DivisionID", divisionId);
        var result = command.ExecuteScalar();
        return result?.ToString() ?? "";
    }

// Handle file selection for both fields
    private async Task OnFileSelected(InputFileChangeEventArgs e, int targetField){
        var file = e.File;
        if (file == null) return;

        // Optional: restrict file types
        var allowedTypes = targetField switch {
            0 => new[] { "image/jpeg", "image/png", "image/gif", "image/bmp" }, // Graphic
            1 => new[] { "application/pdf", "application/msword", "application/vnd.openxmlformats-officedocument.wordprocessingml.document" }, // Spec doc
            _ => Array.Empty<string>()
        };
        if (allowedTypes.Length > 0 && !allowedTypes.Contains(file.ContentType)){
            await JSRuntime.InvokeVoidAsync("alert",
                targetField == 0
                    ? "Please select a valid image file (JPG, PNG, GIF, BMP)"
                    : "Please select a document (PDF, Word)");
            return;
        }

        // Optional: size limit
        if (file.Size > 10 * 1024 * 1024){
            await JSRuntime.InvokeVoidAsync("alert", "File must be less than 10MB");
            return;
        }
        try{
            // Get file name only (like VB6 VBGetOpenFileName returns full path, but we show name)
            var fileName = file.Name;

            // Update the correct field
            switch (targetField){
                case 0: // Graphic Path
                    GraphicPath = fileName;
                    break;
                case 1: // Spec Document
                    SpecDocument = fileName;
                    break;
            }
            mDirty = true;
            StateHasChanged();
        }
        catch (Exception ex){
            await JSRuntime.InvokeVoidAsync("alert", "Error selecting file: " + ex.Message);
        }
    }

    private async Task<dynamic> ShowPickListAsync(string title, List<dynamic> items){
        // Store in component state
        PickListTitle = title;
        PickListItems = items;
        PickListVisible = true;
        StateHasChanged();

        // Wait for user selection
        var tcs = new TaskCompletionSource<dynamic>();
        PickListCallback = (result) => {
            PickListVisible = false;
            tcs.SetResult(result);
            StateHasChanged();
        };
        return await tcs.Task;
    }

    private async Task OnOpenClick(){
        // Step 1: Choose Community
        var communities = await db.QueryAsync(
            "SELECT Area AS Value, Description + ' - ' + Area AS Text FROM tblLocality WHERE Inactive = 0 ORDER BY Description");
        var communityResult = await ShowPickListAsync("Select Community", [communities.ToList()]);
        if (communityResult == null) return;
        string selectedCommunity = communityResult.Value;

        // Step 2: Choose Model (from selected community)
        var models = await db.QueryAsync(@"
        SELECT DISTINCT Model AS Value, Model AS Text 
        FROM tblDBAssemblyMaster 
        WHERE ISNULL(Community,'') = @Community OR Community IS NULL
        ORDER BY Model",
            new { Community = selectedCommunity });
        var modelResult = await ShowPickListAsync("Select Model", [models.ToList()]);
        if (modelResult == null) return;
        string selectedModel = modelResult.Value;

        // Step 3: Choose Option (or Global/DC)
        var options = await db.QueryAsync(@"
        SELECT DISTINCT OptionID AS Value, 
               CASE WHEN OptionID = '' THEN '[Global/DC Option]' ELSE OptionID END AS Text
        FROM tblDBAssemblyMaster 
        WHERE ISNULL(Community,'') = @Community 
          AND ISNULL(Model,'') = @Model
        ORDER BY OptionID",
            new { Community = selectedCommunity, Model = selectedModel });
        var optionResult = await ShowPickListAsync("Select Option (blank = Global)", [options.ToList()]);
        if (optionResult == null) return;
        string selectedOption = optionResult.Value ?? "";

        // Step 4: Choose Assembly
        var assemblies = await db.QueryAsync(@"
        SELECT Assembly AS Value, 
               Assembly + ' - ' + Description AS Text
        FROM tblDBAssemblyMaster 
        WHERE ISNULL(Community,'') = @Community 
          AND ISNULL(Model,'') = @Model 
          AND ISNULL(OptionID,'') = @OptionID
          AND Inactive = 0
        ORDER BY Assembly",
            new {
                Community = selectedCommunity,
                Model = selectedModel,
                OptionID = selectedOption
            });
        if (!assemblies.Any()){
            await JSRuntime.InvokeVoidAsync("alert", "No assemblies found for this selection.");
            return;
        }
        var assemblyResult = await ShowPickListAsync("Select Assembly", [assemblies.ToList()]);
        if (assemblyResult == null) return;
        string selectedAssembly = assemblyResult.Value;

        // Step 5: Load the selected assembly
        mCommunity = selectedCommunity;
        mModel = selectedModel;
        mOptionID = selectedOption;
        mAssembly = selectedAssembly;
        await LoadDataAsync();
    }
