    private async Task LoadDetailsDataAsync(int worksheetId){
        Lines.Clear();
        var sql = @"
            SELECT l.Description AS CommunityDesc, c.Description AS CategoryDesc, 
                   dm.Description AS ModelDescription, d.*
            FROM tblSalesSheetDetails d
            JOIN tblSalesSheetMaster m ON m.Worksheet = d.Worksheet
            LEFT JOIN tblLocality l ON d.Community = l.Area
            LEFT JOIN DivisionCommunities d1 ON d1.Community = l.Area
            LEFT JOIN tblCategories c ON d.Category = c.Category
            LEFT JOIN distinctmodelsbyDivision dm 
                ON dm.DivisionID = m.DivisionID AND d.Model = dm.Model
            WHERE d.Worksheet = @ws
            " + (DivisionID != "" ? "AND (d1.DivisionID IS NULL OR d1.DivisionID = @div)" : "") + @"
            ORDER BY d.Model, d.Assembly";
        var data = await db.QueryAsync(sql, new { ws = worksheetId, div = DivisionID });
        foreach (var row in data){
            var line = new WorksheetLine {
                CommunityDesc = row["CommunityDesc"]?.ToString() ?? "",
                CategoryDesc = row["CategoryDesc"]?.ToString() ?? "",
                ModelDescription = row["ModelDescription"]?.ToString() ?? "",
// ... map all other fields like in VB6 ...
                Description = row["Description"]?.ToString() ?? "",
                Qty = row["Qty"] == null ? 0 : (int)row["Qty"]
// etc.
            };

// Pricing levels mapping (1-10)
            for (int i = 2; i <= 10; i++){
                line.Prices[i] = new PriceLevel {
                    Pretax = row[$"Pretax{i}"] == null ? 0 : (decimal)row[$"Pretax{i}"]
// ... etc.
                };
            }
            Lines.Add(line);
        }

// Always add empty new row
        Lines.Add(new WorksheetLine());
        await MainGrid?.Refresh();
    }

        s = s + "       LEFT OUTER JOIN tblLocality l ON(d.Community=l.Area)" + Environment.NewLine;
        if (!string.IsNullOrEmpty(DivisionID)) // Assume HFApp.DivisionID is string
        {
            s = s + " left outer join DivisionCommunities d1 on d1.Community = l.Area" + Environment.NewLine;
        }
        s = s + "       LEFT OUTER JOIN tblCategories c ON(d.Category=c.Category)" + Environment.NewLine;
        s = s + "       left outer join distinctmodelsbyDivision dm on(dm.DivisionID = m.DivisionID and d.model=dm.model)" + Environment.NewLine;
        s = s + " WHERE d.Worksheet=" + "@Worksheet" + Environment.NewLine;
        if (!string.IsNullOrEmpty(DivisionID)){
            s = s + " and (d1.DivisionID is null or d1.Divisionid = " + "@DivisionID" + ")" + Environment.NewLine;
        }
        s = s + "ORDER BY d.Model,d.Assembly" + Environment.NewLine;
        String mDivisionID = "1";
        parameters = new { Worksheet = mWorksheet };
        var detailResults = await db.QueryAsync(s, parameters);
// .TextMatrix(0, .ColIndex("Community")) = FMain.CD_Community -> Header text; set in <GridColumn HeaderText="@CD_Community">
// .TextMatrix(0, .ColIndex("CommunityPhase")) = FMain.CD_Community & " Phase" -> Similar
// .TextMatrix(0, .ColIndex("CommunityDesc")) = FMain.CD_Community & " Description" -> Similar
        foreach (var row in detailResults){
            r = r + 1;
            var item = new WorksheetDetail();
            if (Convert.ToInt32(row["AssemblyType"]?.ToString() ?? "0") == (int)AssemblyType.atoption){
                item.ModelDescription = row["ModelDescription"]?.ToString() ?? "";
            }

            item.Community = row["Community"]?.ToString() ?? "";
            item.CommunityPhase = row["CommunityPhase"]?.ToString() ?? "";
            item.CommunityDesc = row["CommunityDesc"]?.ToString() ?? "";

            item.AssemblyUOM = row["AssemblyUOM"]?.ToString() ?? "";
            item.Model = row["Model"]?.ToString() ?? "";
            item.Elevation = row["Elevation"]?.ToString() ?? "";
            item.OptionID = row["OptionID"]?.ToString() ?? "";
            item.DCSalesOnly = row["DesignCenterSalesOnly"]?.ToString() ?? "";
            item.SelectByRoom = row["SelectByRoom"]?.ToString() ?? "";
            item.DisplayTotalOnly = row["DisplayTotalOnly"]?.ToString() ?? "";
            item.JCExtra = row["JCExtra"]?.ToString() ?? "";
            item.Series = row["Series"]?.ToString() ?? "";
            item.FloorArea = Convert.ToDouble(row["FloorArea"]?.ToString() ?? "0");

            item.Bathrooms = row["Bathrooms"]?.ToString() ?? "";
            item.Style = row["Style"]?.ToString() ?? "";

            item.SpecDocument = row["SpecDocument"]?.ToString() ?? "";
            item.GraphicPath = row["GraphicPath"]?.ToString() ?? "";
            item.MaxWidth = Convert.ToDouble(row["MaxWidth"]?.ToString() ?? "0");
            item.MaxLength = Convert.ToDouble(row["MaxLength"]?.ToString() ?? "0");
            item.Inactive = (row["Inactive"]?.ToString() == "True");
            item.Color = row["Color"]?.ToString() ?? "";
            item.Location = row["Location"]?.ToString() ?? "";
            item.Qty = row["Qty"]?.ToString() ?? "";
            if (item.Qty == "0") item.Qty = "1";
            item.IncludedOption = row["IncludedOption"]?.ToString() ?? "";
            item.ReadyToPublish = (row["ReadyToPublish"]?.ToString() == "True");
            item.ColorListID = row["ColorListID"]?.ToString() ?? "";
            item.StyleListID = row["StyleListID"]?.ToString() ?? "";
            item.FinishListID = row["FinishListID"]?.ToString() ?? "";
            item.OtherListID = row["OtherListID"]?.ToString() ?? "";
            item.StyleValue = row["StyleValue"]?.ToString() ?? "";
            item.FinishValue = row["FinishValue"]?.ToString() ?? "";
            item.OtherValue = row["OtherValue"]?.ToString() ?? "";
            item.Description = row["Description"]?.ToString() ?? "";
            item.Notes = row["Notes"]?.ToString() ?? "";
            item.Comments = row["Comments"]?.ToString() ?? "";
            item.AssemblyType = row["AssemblyType"]?.ToString() ?? "";
            item.Category = row["Category"]?.ToString() ?? "";
            item.CategoryDesc = row["CategoryDesc"]?.ToString() ?? "";
            item.Markup = Convert.ToDouble(row["Markup"]?.ToString() ?? "0");
            item.Margin = Convert.ToDouble(row["Margin"]?.ToString() ?? "0");
            item.COMarkup = Convert.ToDouble(row["COMarkup"]?.ToString() ?? "0");
            item.COMargin = Convert.ToDouble(row["COMargin"]?.ToString() ?? "0");
            item.RoundTo = Convert.ToDouble(row["Roundto"]?.ToString() ?? "0");
            item.LandCost = Convert.ToDouble(row["LandCost"]?.ToString() ?? "0");
            item.IncentiveRetail = Convert.ToDouble(row["IncentiveRetail"]?.ToString() ?? "0");
            item.IncentiveCost = Convert.ToDouble(row["IncentiveCost"]?.ToString() ?? "0");
            item.ConstructionCost = Convert.ToDouble(row["ConstructionCost"]?.ToString() ?? "0");
            item.Cost = Convert.ToDouble(row["Cost"]?.ToString() ?? "0");
            if (!TaxIncl){
                item.Pretax = RoundToPrice(Convert.ToDouble(row["Pretax"]?.ToString() ?? "0"), Convert.ToDouble(row["Roundto"]?.ToString() ?? "0"));
            }
            else{
                item.Pretax = Convert.ToDouble(row["Pretax"]?.ToString() ?? "0");
            }
            item.Tax = Convert.ToDouble(row["Tax"]?.ToString() ?? "0");
            item.Total = item.Pretax + item.Tax;
            if (!TaxIncl){
                item.COPretax = RoundToPrice(Convert.ToDouble(row["COPretax"]?.ToString() ?? "0"), Convert.ToDouble(row["Roundto"]?.ToString() ?? "0"));
            }
            else{
                item.COPretax = Convert.ToDouble(row["COPretax"]?.ToString() ?? "0");
            }
            item.COTax = Convert.ToDouble(row["COTax"]?.ToString() ?? "0");
            item.COTotal = item.COPretax + item.COTax;
            item.InSpec = (row["IncludedInSpec"]?.ToString() == "True");
            for (i = 2; i <= 10; i++){
                if (!TaxIncl){
                    typeof(WorksheetDetail).GetProperty($"Pretax{i}").SetValue(item, RoundToPrice(Convert.ToDouble(row[$"Pretax{i}"]?.ToString() ?? "0"), Convert.ToDouble(row["Roundto"]?.ToString() ?? "0")));
                }
                else{
                    typeof(WorksheetDetail).GetProperty($"Pretax{i}").SetValue(item, Convert.ToDouble(row[$"Pretax{i}"]?.ToString() ?? "0"));
                }
                typeof(WorksheetDetail).GetProperty($"Tax{i}").SetValue(item, Convert.ToDouble(row[$"Tax{i}"]?.ToString() ?? "0"));

                typeof(WorksheetDetail).GetProperty($"Profit{i}").SetValue(item, (double)typeof(WorksheetDetail).GetProperty($"Pretax{i}").GetValue(item) - item.Cost);
            }
            item.IncludeTax = (row["IncludeTax"]?.ToString() == "True");
            item.Profit = item.Pretax - item.Cost;
            if (item.FloorArea == 0){
                item.CostPerFt = 0; // Empty equivalent
                item.PricePerFt = 0; // Empty equivalent
            }
            else{
                item.CostPerFt = item.ConstructionCost / item.FloorArea;
                item.PricePerFt = item.Pretax / item.FloorArea;
            }
            GridData.Add(item);
        }

        mDirty = false;
    }
