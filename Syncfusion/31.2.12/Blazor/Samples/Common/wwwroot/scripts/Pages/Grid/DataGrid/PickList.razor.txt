    private List<string> KeyColumnList = new();
    private List<string> HiddenColumnList = new();

    public class ViewItem { public int Id { get; set; } public string Name { get; set; } = ""; }
    public class ColumnConfig
    {
        public string Field { get; set; } = "";
        public string Header { get; set; } = "";
        public int Width { get; set; } = 120;
        public bool Visible { get; set; } = true;
    }

    protected override async Task OnInitializedAsync()
    {
        mTitle = Title;
        mSQL = Sql;

        // Load persisted fast load setting
        chkFast = true;

        // Load views
        if (!string.IsNullOrEmpty(Views))
        {
            var names = Views.Split('|', StringSplitOptions.RemoveEmptyEntries);
            ViewItems = names.Select((n, i) => new ViewItem { Id = i + 1, Name = n.Trim() }).ToList();
            viewIndex =  1;
        }

        // Parse key/hidden columns
        KeyColumnList = KeyColumns?.Split(',', StringSplitOptions.RemoveEmptyEntries).Select(s => s.Trim()).ToList() ?? new();
        HiddenColumnList = HiddenColumns?.Split(',', StringSplitOptions.RemoveEmptyEntries).Select(s => s.Trim()).ToList() ?? new();

        await LoadDataAndBuildColumns();
        //await LoadSavedColumnState();
    }

    private DbWrapperSqlServer db = new();

    private async Task LoadDataAndBuildColumns()
    {
        //var data = await db.QueryAsync(mSQL);
        //GridData = data.Select(r => (ExpandoObject)r).ToList();
        GridData = await db.QueryDynamicAsync(mSQL);


        if (!GridData.Any()) return;

        var first = (IDictionary<string, object>)GridData[0];
        var fields = first.Keys.ToList();

        ColumnsConfig.Clear();

        if (!string.IsNullOrEmpty(Columns))
        {
            var headers = Columns.Split('|');
            for (int i = 0; i < headers.Length && i < fields.Count; i++)
            {
                var header = headers[i].TrimStart('<');
                var field = fields[i];
                ColumnsConfig.Add(new ColumnConfig
                {
                    Field = field,
                    Header = header,
                    Width = 120,
                    Visible = !HiddenColumnList.Contains(field)
                });
            }
        }
        else
        {
            foreach (var field in fields)
            {
                ColumnsConfig.Add(new ColumnConfig
                {
                    Field = field,
                    Header = field,
                    Width = 120,
                    Visible = !HiddenColumnList.Contains(field)
                });
            }
        }

        StateHasChanged();
    }

    private async Task LoadSavedColumnState()
    {
        var saved = await JSRuntime.InvokeAsync<string>("localStorage.getItem", $"{mTitle}_View{viewIndex}_Columns");
        if (string.IsNullOrEmpty(saved)) return;

        foreach (var part in saved.Split('|', StringSplitOptions.RemoveEmptyEntries))
        {
            var kv = part.Split('=');
            if (kv.Length != 2) continue;

            var col = ColumnsConfig.FirstOrDefault(c => c.Field == kv[0]);
            if (col == null) continue;

            if (kv[1].StartsWith("w") && int.TryParse(kv[1].AsSpan(1), out int w))
                col.Width = w;
            else if (kv[1] == "hidden")
                col.Visible = false;
        }

        if (grid != null)
            await grid.RefreshColumnsAsync();
    }

    private async Task SaveColumnState()
    {
        var parts = new List<string>();
        var columns = await grid.GetColumnsAsync();

        foreach (var col in columns)
        {
            parts.Add(col.Visible ? $"{col.Field}=w{col.Width}" : $"{col.Field}=hidden");
        }

        var state = string.Join("|", parts);
        await JSRuntime.InvokeVoidAsync("localStorage.setItem", $"{mTitle}_View{viewIndex}_Columns", state);
    }

    private async Task OnOkClick()
    {
        Console.WriteLine("OnClick"); 
        var selected = await grid.GetSelectedRecordsAsync();
        if (selected == null || selected.Count == 0) return;

        var result = selected.Cast<IDictionary<string, object>>().ToList();
        //_navigationManager.NavigateTo("/HF");
        await OnSelected.InvokeAsync(result);
        Navigation.NavigateTo(CancelNavigateTo);
    }

    private async Task OnFastLoadChanged(ChangeEventArgs<bool> args)
    {
        if (mLoading) return;

    }
