    // === LOAD PRICING COMMUNITIES ===
    private async Task LoadPricingCommunitiesAsync(string? limitToCommunity = null) {
        var limitClause = string.IsNullOrEmpty(limitToCommunity) ? "" : "AND c.area = @LimitToCommunity";
        var sql = $@"
            SELECT c.Description, c.area + CHAR(2) AS Value, 0 AS SortOrder
            FROM tbllocality c
            LEFT OUTER JOIN DivisionCommunities d ON d.Community = c.Area AND d.DivisionID = @DivisionID
            WHERE ISNULL(c.inactive, 0) = 0
              AND (@DivisionID = 0 OR d.DivisionID = @DivisionID)
              {limitClause}
            UNION
            SELECT c.Description + ISNULL(' ' + p.description, ''), 
                   c.area + CHAR(2) + ISNULL(p.communityphase, '') AS Value, 0 AS SortOrder
            FROM tbllocality c
            LEFT OUTER JOIN communityphase p ON c.area = p.community
            LEFT OUTER JOIN DivisionCommunities d ON d.Community = c.Area AND d.DivisionID = @DivisionID
            WHERE ISNULL(c.inactive, 0) = 0
              AND (@DivisionID = 0 OR d.DivisionID = @DivisionID)
              {limitClause}
            ORDER BY 1";
        
        var rows = await db.QueryAsync(sql, new { 
            DivisionID = AppState_DivisionID, 
            LimitToCommunity = limitToCommunity 
        });
        
        PricingCommunities = rows.Select(row => new CommunityItem {
            PricingCommunityPhase = row["Description"]?.ToString() ?? "",
            PricingCommunity = row["Value"]?.ToString() ?? "",
            SortOrder = Convert.ToInt32(row["SortOrder"])
        }).OrderBy(x => x.PricingCommunityPhase).ToList();

        PricingCommunities.Insert(0, new CommunityItem { 
            PricingCommunityPhase = "", 
            PricingCommunity = "", 
            SortOrder = 0 
        });
    }

    private void UpdateUIVisibility() {
        switch (mAssemblyType) {
            case AssemblyTypes.atModel:
                ShowOptionField = false;
                ShowModelField = true;
                ShowSeriesFields = true;
                ShowModelFields = true;
                ShowOptionFields = false;
                ModelOptionLabel = "Model";
                break;
            case AssemblyTypes.atoption:
                ShowOptionField = true;
                ShowModelField = true;
                ShowSeriesFields = true;
                ShowModelFields = false;
                ShowOptionFields = true;
                ModelOptionLabel = "Model";
                break;
            case AssemblyTypes.atGlobal:
            case AssemblyTypes.atDesignCenter:
            case AssemblyTypes.atCustomOption:
                ShowOptionField = true;
                ShowModelField = false;
                ShowSeriesFields = false;
                ShowModelFields = false;
                ShowOptionFields = true;
                ModelOptionLabel = "Option";
                break;
        }
        UOMEnabled = mAssemblyType != AssemblyTypes.atModel;
    }

    // === LOAD ITEMS GRID ===
    private async Task LoadItemsGridAsync() {
        var itemsQuery = string.IsNullOrEmpty(PricingCommunity) 
            ? BuildItemsQuery_NoPricing() 
            : BuildItemsQuery_WithPricing();

        var itemsData = await db.QueryAsync(itemsQuery, new {
            DivisionID = AppState_DivisionID,
            Community = mCommunity?.Trim() ?? "",
            Model = mModel?.Trim() ?? "",
            OptionID = mOptionID?.Trim() ?? "",
            Assembly = mAssembly?.Trim() ?? "",
            PricingCommunity = mPricingCommunity?.Trim() ?? "",
            PricingCommunityPhase = mPricingCommunityPhase?.Trim() ?? ""
        });

        var items = new List<AssemblyItemRow>();
        AssemblyCost = 0;

        foreach (var row in itemsData) {
            var item = new AssemblyItemRow {
                Sequence = ToInt(row["sequence"]),
                Phase = row["Phase"]?.ToString() ?? "",
                Item = row["Item"]?.ToString() ?? "",
                ItemNumber = row["itemnumber"]?.ToString() ?? "",
                ItemChart = row["itemchart"]?.ToString() ?? "",
                Description = row["Description"]?.ToString() ?? "",
                TakeoffUOM = row["TakeoffUOM"]?.ToString() ?? "",
                OrderUOM = row["OrderUOM"]?.ToString() ?? "",
                Formula = row["Formula"]?.ToString() ?? "",
                TakeoffQty = ToDouble(row["TakeoffQty"]),
                OrderQty = ToDouble(row["OrderQty"]),
                WastePercent = ToDouble(row["WastePercent"]),
                Notes = row["notes"]?.ToString() ?? "",
                Location = row["Location"]?.ToString() ?? "",
                IsQuote = row["IsQuote"]?.ToString() == "True",
                Invertable = row["Invertable"]?.ToString() == "True" && ToInt(row["hasInverseItem"]) == 1,
                UseModelCost = row["UseModelCost"]?.ToString() == "True",
                POIndex = row["POIndex"]?.ToString() ?? "",
                POIndexDescription = row["POIndexDescription"]?.ToString() ?? "",
                JCCostCode = row["JCCostCode"]?.ToString() ?? "",
                JCCategory = row["JCCategory"]?.ToString() ?? ""
            };

            // WBS fields
            for (int i = 1; i <= 40; i++) {
                item.WBS[i] = row[$"WBS{i:D2}"]?.ToString() ?? "";
            }

            if (!string.IsNullOrEmpty(PricingCommunity)) {
                var quality = ToInt(row["RateQuality"]);
                item.PriceLevel = quality switch {
                    4 or 5 or 6 => "Quote",
                    9 => "Phase Specific",
                    10 => "Community Specific",
                    11 => "Global (any Community)",
                    12 => "Corporate (any Division)",
                    13 => "Item DB",
                    _ => "Unit Price"
                };
                item.Price = ToDouble(row["Rate"]);
                item.TaxRate = ToDouble(row["JCTaxRate"]) / 100;
                item.PretaxAmount = item.Price * item.OrderQty;
                item.TaxAmount = item.PretaxAmount * item.TaxRate;
                item.ExtendedAmount = item.PretaxAmount + item.TaxAmount;
                AssemblyCost += item.ExtendedAmount;
                item.Vendor = row["Vendor"]?.ToString() ?? "";
                item.VendorDesc = row["VendorDesc"]?.ToString() ?? "";
                item.OldPriceLevel = item.PriceLevel;
            } else {
                item.ItemType = item.IsQuote ? "Quote" : "Unit Price";
            }

            items.Add(item);
        }

        ItemsData = items;
        AssemblyCostLabel = AssemblyCost.ToString("$#,##0.00");
    }

    // === LOAD COMPONENTS GRID ===
    private async Task LoadComponentsGridAsync() {
        var checkData = await db.QueryAsync(
            "SELECT parentassemblyid FROM tblDBAssemblyComponents WHERE ComponentAssemblyID = @Id",
            new { Id = mAssemblyID });
        
        var isChildComponent = checkData.Any();
        ShowComponentsGrid = !isChildComponent && mUseComponents;

        if (!ShowComponentsGrid) {
            ComponentsData = new List<AssemblyComponentRow>();
            return;
        }

        var compQuery = @"
            SELECT m.assemblyid, m.community, m.assembly, m.model, m.optionid, m.description, c.qty,
                   ROUND(SUM(
                       ISNULL(dbo.Purch_GetItemRate(0,0,@PricingCommunity,@PricingCommunityPhase,d.Assembly,d.Model,d.OptionID,d.Phase,d.Item,d.Sequence,v.Vendor_id,GETDATE(),@DivisionID),0)
                       * (100 + ISNULL(t.JCRate,0)) / 100
                       * ISNULL(d.OrderQty,0)
                   ), 2) AS Cost
            FROM tblDBAssemblyComponents c
            JOIN tblDBAssemblyMaster m ON c.componentassemblyid = m.assemblyid
            LEFT OUTER JOIN tblDBAssemblyDetails d ON m.assemblyid = d.assemblyid
            LEFT OUTER JOIN tblPhaseItem i ON d.DivisionID = i.DivisionID AND d.Phase = i.Phase AND d.Item = i.Item
            LEFT OUTER JOIN tblPOIndex p ON p.DivisionID = @DivisionID AND ISNULL(NULLIF(d.POIndex,''), ISNULL(i.POIndex,'')) = p.POIndex
            LEFT OUTER JOIN tblVendors v ON v.DivisionID = d.DivisionID AND v.vendor_id = dbo.Purch_GetCommunityVendor(@PricingCommunity, p.POIndex, @DivisionID)
            LEFT OUTER JOIN taxGroups t ON t.DivisionID = @DivisionID AND t.TaxGroup = dbo.Purch_GetDefaultTaxGroup('', @PricingCommunity, '', d.Model, d.Assembly, i.phase, i.item, v.Vendor_ID, i.JCCategory, @DivisionID)
            WHERE c.ParentAssemblyid = @ParentAssemblyId
            GROUP BY m.assemblyid, m.community, m.assembly, m.model, m.optionid, m.description, c.qty";

        var compData = await db.QueryAsync(compQuery, new {
            ParentAssemblyId = mAssemblyID,
            DivisionID = AppState_DivisionID,
            PricingCommunity = mPricingCommunity ?? "",
            PricingCommunityPhase = mPricingCommunityPhase ?? ""
        });

        var components = new List<AssemblyComponentRow>();
        ComponentCost = 0;

        foreach (var row in compData) {
            var comp = new AssemblyComponentRow {
                AssemblyId = Convert.ToInt64(row["assemblyid"]),
                Community = row["community"]?.ToString() ?? "",
                Model = row["model"]?.ToString() ?? "",
                Option = row["optionid"]?.ToString() ?? "",
                Assembly = row["assembly"]?.ToString() ?? "",
                Description = row["description"]?.ToString() ?? "",
                Qty = ToDouble(row["qty"]),
                Cost = !string.IsNullOrEmpty(PricingCommunity) ? ToDouble(row["Cost"]) : 0
            };
            ComponentCost += comp.Cost * comp.Qty;
            components.Add(comp);
        }

        ComponentsData = components;
    }

    // === SAVE DATA ===
    private async Task<bool> SaveDataAsync() {
        if (!ValidateData()) return false;

        try {
            // Insert or Update Master
            if (mAssemblyID == 0) {
                await InsertAssemblyMasterAsync();
            } else {
                await UpdateAssemblyMasterAsync();
            }

            // Save Items
            await SaveItemsAsync();

            // Save Components
            await SaveComponentsAsync();

            mDirty = false;

    private async Task SaveItemsAsync() {
        foreach (var item in ItemsData) {
            if (item.RowStatus == "delete" && item.Sequence != 0) {
                await db.ExecuteAsync(
                    "DELETE FROM tblDBAssemblyDetails WHERE Sequence = @Sequence",
                    new { item.Sequence });
            } else if (item.RowStatus == "new") {
                await db.ExecuteAsync(@"
                    INSERT INTO tblDBAssemblyDetails(
                        AssemblyID, DivisionID, Community, Assembly, Model, OptionID, Phase, Item,
                        ItemChart, Notes, UStmp, TStmp, Formula, TakeoffQty, OrderQty, POIndex,
                        Location, UseModelCost, Invertable, ConversionFactor
                    ) VALUES (
                        @AssemblyID, @DivisionID, @Community, @Assembly, @Model, @OptionID, @Phase, @Item,
                        @ItemChart, @Notes, @UStmp, GETDATE(), @Formula, @TakeoffQty, @OrderQty, @POIndex,
                        @Location, @UseModelCost, @Invertable, @ConversionFactor
                    )", new {
                        AssemblyID = mAssemblyID,
                        DivisionID = AppState_DivisionID,
                        Community = CommunityValue ?? "",
                        Assembly = AssemblyCode ?? "",
                        Model = ModelValue ?? "",
                        OptionID = OptionID ?? "",
                        item.Phase,
                        item.Item,
                        item.ItemChart,
                        item.Notes,
                        UStmp = "system",
                        item.Formula,
                        item.TakeoffQty,
                        item.OrderQty,
                        item.POIndex,
                        item.Location,
                        item.UseModelCost,
                        item.Invertable,
                        ConversionFactor = item.AssemblyConversionFactor
                    });
            } else if (item.RowStatus == "update") {
                await db.ExecuteAsync(@"
                    UPDATE tblDBAssemblyDetails SET
                        Phase = @Phase, Item = @Item, ItemChart = @ItemChart, Notes = @Notes,
                        UStmp = @UStmp, TStmp = GETDATE(), Formula = @Formula, TakeoffQty = @TakeoffQty,
                        OrderQty = @OrderQty, POIndex = @POIndex, Location = @Location,
                        UseModelCost = @UseModelCost, Invertable = @Invertable, ConversionFactor = @ConversionFactor
                    WHERE Sequence = @Sequence", new {
                        item.Sequence,
                        item.Phase,
                        item.Item,
                        item.ItemChart,
                        item.Notes,
                        UStmp = "system",
                        item.Formula,
                        item.TakeoffQty,
                        item.OrderQty,
                        item.POIndex,
                        item.Location,
                        item.UseModelCost,
                        item.Invertable,
                        ConversionFactor = item.AssemblyConversionFactor
                    });
            }
        }
    }

    private async Task SaveComponentsAsync() {
        foreach (var comp in ComponentsData) {
            if (comp.RowStatus == "delete") {
                await db.ExecuteAsync(
                    "DELETE FROM tblDBAssemblyComponents WHERE ParentAssemblyId = @ParentId AND ComponentAssemblyId = @CompId",
                    new { ParentId = mAssemblyID, CompId = comp.AssemblyId });
            } else if (comp.RowStatus == "new") {
                await db.ExecuteAsync(@"
                    INSERT INTO tblDBAssemblyComponents(ParentAssemblyId, ComponentAssemblyId, Qty)
                    VALUES (@ParentId, @CompId, @Qty)",
                    new { ParentId = mAssemblyID, CompId = comp.AssemblyId, comp.Qty });
            }
        }
    }

    private bool ValidateData() {
        if (string.IsNullOrWhiteSpace(AssemblyCode)) {
            JSRuntime.InvokeVoidAsync("alert", "Assembly code is required");
            return false;
        }
        if (string.IsNullOrWhiteSpace(Description)) {
            JSRuntime.InvokeVoidAsync("alert", "Description is required");
            return false;
        }
        return true;
    }

    private async void OnPriceCommunityChanged(Syncfusion.Blazor.DropDowns.ChangeEventArgs<string, CommunityItem> args) {
        SelectedPriceCommunity = args.Value;
        if (!string.IsNullOrEmpty(SelectedPriceCommunity)) {
            var parts = SelectedPriceCommunity.Split((char)2);
            mPricingCommunity = parts[0];
            mPricingCommunityPhase = parts.Length > 1 ? parts[1] : "";
        } else {
            mPricingCommunity = "";
            mPricingCommunityPhase = "";
        }
        PricingCommunity = mPricingCommunity;
        PricingCommunityPhase = mPricingCommunityPhase;
        await LoadDataAsync();
        StateHasChanged();
    }

    private async Task OnFileSelected(InputFileChangeEventArgs e, int targetField) {
        var file = e.File;
        if (file == null) return;

        switch (targetField) {
            case 0:
                GraphicPath = file.Name;
                break;
            case 1:
                SpecDocument = file.Name;
                break;
        }
        mDirty = true;
        StateHasChanged();
    }

    private async Task<bool> ConfirmSaveAsync() {
        if (!mDirty) return true;
        var result = await JSRuntime.InvokeAsync<bool>("confirm", "Save changes?");
        if (result) await SaveDataAsync();
        return true;
    }

    private static int ToInt(object? val, int defaultValue = 0) {
        if (val == null || val is DBNull) return defaultValue;
        return int.TryParse(val.ToString(), out var result) ? result : defaultValue;
    }

    private static double ToDouble(object? val, double defaultValue = 0) {
        if (val == null || val is DBNull) return defaultValue;
        return double.TryParse(val.ToString(), out var result) ? result : defaultValue;
    }
