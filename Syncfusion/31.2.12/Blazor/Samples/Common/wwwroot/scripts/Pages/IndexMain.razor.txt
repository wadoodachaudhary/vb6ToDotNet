@code {
    [Inject] NavigationManager NavigationManager { get; set; } = default!;

    private void NavigateTo(string taskName)
    {
        // Map task names to routes or run equivalent logic from VB6 Task_MouseDown
        switch (taskName)
        {
            case "Salespricing Worksheets":
                NavigationManager.NavigateTo("/estimating"); // Example route
                break;
            case "Model & Option Library":
                NavigationManager.NavigateTo("/picklist3"); // Route to your Model/Option page
                break;
            case "Communities":
                NavigationManager.NavigateTo("/communities");
                break;
            case "Vendor Pricelists":
                NavigationManager.NavigateTo("/vendor-pricelists");
                break;
            case "Vendor":
                NavigationManager.NavigateTo("/vendors");
                break;
            case "Default Vendors":
                NavigationManager.NavigateTo("/default-vendors");
                break;
            case "Item DB":
                NavigationManager.NavigateTo("/itemdb");
                break;
            // Add more as needed
            default:
                break;
        }
    }
}
